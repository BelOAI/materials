\section*{Задачи}

\begin{exercise}
Вычислите производную функции $f(x) = x^2 \ln x$ при $x > 0$.
\answerlines{4}
\end{exercise}

\begin{exercise}
Решите уравнение $2^x = 8$.
\answerbox[2.5cm]
\end{exercise}

\begin{exercise}
Докажите, что для любых вещественных $a$ и $b$ выполняется неравенство
\[
  (a + b)^2 \geq 4ab.
\]
\answerlines{6}
\end{exercise}
