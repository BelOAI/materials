\begin{frame}{Формула и ключевые числа}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \lead{Пример формулы}
      \vspace{0.5em}
      \[
        S = 0{,}30\,S_1 + 0{,}50\,S_2 + 0{,}20\,S_3
      \]
      \vspace{-0.5em}
      \begin{itemize}\setlength\itemsep{0.3em}
        \item $S_1$ --- первый компонент;
        \item $S_2$ --- второй компонент;
        \item $S_3$ --- третий компонент.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Ключевые числа}
        \begin{tabular}{@{}p{0.62\textwidth}>{\raggedleft\arraybackslash}p{0.26\textwidth}@{}}
          Участников на этап & \textbf{не более 50} \\[0.35em]
          Проходят дальше & \textbf{не более 6} \\[0.35em]
          Резервный состав & \textbf{20} \\
        \end{tabular}
      \end{block}
      {\footnotesize\color{acGray}
      \tbd{Черновик} --- пометка для данных, которые ещё не утверждены.}
    \end{column}
  \end{columns}
\end{frame}
