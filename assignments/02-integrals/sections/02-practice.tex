\begin{enumerate}
\item $\int(x^2-2x+1)\,dx$
\item $\int(3x^2-2\cos x)\,dx$

\item $\int\dfrac{2x^2-8}{16-x^4}\,dx$
\item $\int3xe^{2x}\,dx$
\end{enumerate}
\end{multicols}
