\section*{Практикум 1. Алгебраические пределы и раскрытие неопределённостей $\left[\frac00\right]$}
\textit{Указание: разложите числитель и знаменатель на множители или домножьте на сопряжённое выражение.}
\begin{multicols}{2}
\begin{enumerate}
\item $\lim\limits_{x\to0}\dfrac{3x^2-2x}{2x^2-5x}=\dfrac25$
\item $\lim\limits_{x\to2}\dfrac{x^2-7x+10}{8-x^3}=\dfrac14$
\item $\lim\limits_{x\to1/2}\dfrac{8x^3-1}{6x^2-5x+1}=6$
\item $\lim\limits_{x\to2}\dfrac{x^3-3x-2}{x^3-8}=\dfrac34$
\item $\lim\limits_{x\to2}\dfrac{x^2-5x+6}{x^3-2x^2-x+2}=-\dfrac13$
\item $\lim\limits_{x\to-1}\dfrac{3x^2+2x-1}{2+x-x^2}=-\dfrac43$
\item $\lim\limits_{x\to-2}\left(\dfrac1{x+2}-\dfrac{12}{x^3+8}\right)=-\dfrac12$
\item $\lim\limits_{x\to-1}\dfrac{x^2+3x-1}{x^3+1}=\infty$
\item $\lim\limits_{x\to0}\dfrac{4x^2-3x}{2x^2-9x}=\dfrac13$
\item $\lim\limits_{x\to1}\dfrac{x-1}{\sqrt{x}-1}=2$
\item $\lim\limits_{x\to0}\dfrac{\sqrt[3]{1-x}-1}{x}=-\dfrac13$
\item $\lim\limits_{x\to3}\dfrac{\sqrt{x+13}-4}{x^2-9}=\dfrac1{48}$
\item $\lim\limits_{x\to1}\dfrac{\sqrt{5-x}-2}{\sqrt{2-x}-1}=\dfrac12$
\item $\lim\limits_{x\to0}\dfrac{x^2}{\sqrt{x^2+4}-2}=4$
\item $\lim\limits_{x\to4}\dfrac{2-\sqrt{x}}{\sqrt{6x+1}-5}=-\dfrac5{12}$
\item $\lim\limits_{x\to0}\dfrac{x^2-x}{\sqrt{x}-1}=0$
\item $\lim\limits_{x\to0}\dfrac{x}{\sqrt{5-x}-\sqrt{5+x}}=-\sqrt5$
\item $\lim\limits_{x\to3}\dfrac{\sqrt{x-2}-1}{x-3}=\dfrac12$
\item $\lim\limits_{x\to2}\dfrac{\sqrt{x+7}-3}{\sqrt{x+2}-2}=\dfrac23$
\item $\lim\limits_{x\to1}\dfrac{\sqrt[3]{x}-1}{\sqrt{x}-\sqrt{3x}}=2$
\item $\lim\limits_{x\to-1}\dfrac{\sqrt[3]{x+1}}{x+1}=\dfrac13$
\item $\lim\limits_{x\to5}\dfrac{\sqrt{x-1}-2}{x-5}=\dfrac14$
\end{enumerate}
\end{multicols}

\section*{Практикум 2. Первый замечательный предел}
\begin{multicols}{2}
\begin{enumerate}
\item $\lim\limits_{x\to0}\dfrac{\sin2x}{3x}=\dfrac23$
\item $\lim\limits_{x\to0}\dfrac{\sin2x+\sin3x}{x}=5$
\item $\lim\limits_{x\to0}\dfrac{\sin3x}{\sin2x}=\dfrac32$
\item $\lim\limits_{x\to0}\dfrac{\sqrt{x+4}-2}{\sin5x}=\dfrac1{20}$
\item $\lim\limits_{x\to0}\dfrac{1-\cos6x}{x\sin3x}=6$
\item $\lim\limits_{x\to0}\dfrac{\cos4x-\cos2x}{2x^2}=-3$
\item $\lim\limits_{x\to0}\dfrac{1-\cos5x}{2x^2}=\dfrac{25}{4}$
\item $\lim\limits_{x\to0}\dfrac{\sin4x}{\sqrt{x+1}-1}=8$
\item $\lim\limits_{x\to0}\dfrac{\sin7x-\sin2x}{\sin x}=5$
\item $\lim\limits_{x\to0}\dfrac{1-\cos x}{x\sin x}=\dfrac12$
\item $\lim\limits_{x\to0}\dfrac{\sin2x}{x}=2$
\item $\lim\limits_{x\to0}\dfrac{\sin^22x}{x^2}=4$
\item $\lim\limits_{x\to0}\dfrac{\sin2x\sin x}{x^2}=2$
\item $\lim\limits_{x\to0}\dfrac{\sin3x}{3\tg x}=1$
\item $\lim\limits_{x\to0}\dfrac{\sin x}{\sin7x-\sin5x}=\dfrac12$
\item $\lim\limits_{x\to0}\dfrac{\sin2x}{x^2+6x}=\dfrac13$
\item $\lim\limits_{x\to0}\dfrac{\cos^2x-\cos^22x}{3x^2}=1$
\item $\lim\limits_{x\to0}\left(\dfrac1{\sin2x}-\dfrac1{\tg2x}\right)=0$
\item $\lim\limits_{x\to0}\dfrac{\cos2x-1}{x\sin x}=-2$
\item $\lim\limits_{x\to0}\dfrac{1-\cos3x}{\sin^27x}=\dfrac9{98}$
\item $\lim\limits_{x\to0}\dfrac{\sqrt{1+\sin x}-\sqrt{1-\sin x}}{\tg x}=1$
\end{enumerate}
\end{multicols}

\section*{Практикум 3. Пределы на бесконечности и второй замечательный предел}
\begin{multicols}{2}
\begin{enumerate}
\item $\lim\limits_{x\to\infty}\dfrac{x^4-2x^2+3}{3x^3-5}=\infty$
\item $\lim\limits_{x\to\infty}\dfrac{3x^2+x+1}{5x^2-2x-3}=\dfrac35$
\item $\lim\limits_{x\to\infty}\dfrac{\sqrt{x^2+1}+x}{\sqrt[4]{x^3+x}-x}=-1$
\item $\lim\limits_{x\to\infty}\dfrac{(x+1)^3-(x-1)^3}{(x+1)^2+(x-1)^2}=3$
\item $\lim\limits_{x\to\infty}(\sqrt{x^2+x-1}-\sqrt{x^2-x+1})=1$
\item $\lim\limits_{x\to0}(1+x)^{1/x}=e$
\item $\lim\limits_{x\to\infty}\left(\dfrac{3x-2}{3x+5}\right)^{3x+1}=e^{-7}$
\item $\lim\limits_{x\to+\infty}\left(\dfrac{1+x}{2+x}\right)^{1-x}=1$
\end{enumerate}
\end{multicols}
