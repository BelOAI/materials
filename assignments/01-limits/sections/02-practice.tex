\section*{Практикум 1. Алгебраические пределы}
\textit{Сначала проверьте подстановку и область определения. При неопределённости разложите на множители или домножьте на сопряжённое. Если односторонние пределы различаются, двустороннего предела нет.}
\begin{multicols}{2}
\begin{enumerate}
\item$\lim\limits_{x\to0}\dfrac{3x^2-2x}{2x^2-5x}$
\item$\lim\limits_{x\to2}\dfrac{x^2-7x+10}{8-x^3}$
\item$\lim\limits_{x\to1/2}\dfrac{8x^3-1}{6x^2-5x+1}$
\item$\lim\limits_{x\to2}\dfrac{x^3-3x-2}{x^3-8}$
\item$\lim\limits_{x\to2}\dfrac{x^2-5x+6}{x^3-2x^2-x+2}$
\item$\lim\limits_{x\to-1}\dfrac{3x^2+2x-1}{2+x-x^2}$
\item$\lim\limits_{x\to-2}\left(\dfrac1{x+2}-\dfrac{12}{x^3+8}\right)$
\item$\lim\limits_{x\to-1}\dfrac{x^2+3x-1}{x^3+1}$
\item$\lim\limits_{x\to0}\dfrac{4x^2-3x}{2x^2-9x}$
\item$\lim\limits_{x\to1}\dfrac{x-1}{\sqrt{x}-1}$
\item$\lim\limits_{x\to0}\dfrac{\sqrt[3]{1-x}-1}{x}$
\item$\lim\limits_{x\to3}\dfrac{\sqrt{x+13}-4}{x^2-9}$
\item$\lim\limits_{x\to1}\dfrac{\sqrt{5-x}-2}{\sqrt{2-x}-1}$
\item$\lim\limits_{x\to0}\dfrac{x^2}{\sqrt{x^2+4}-2}$
\item$\lim\limits_{x\to4}\dfrac{2-\sqrt{x}}{\sqrt{6x+1}-5}$
\item$\lim\limits_{x\to0}\dfrac{x^2-x}{\sqrt{x}-1}$
\item$\lim\limits_{x\to0}\dfrac{x}{\sqrt{5-x}-\sqrt{5+x}}$
\item$\lim\limits_{x\to3}\dfrac{\sqrt{x-2}-1}{x-3}$
\item$\lim\limits_{x\to2}\dfrac{\sqrt{x+7}-3}{\sqrt{x+2}-2}$
\item$\lim\limits_{x\to1}\dfrac{\sqrt[3]{x}-1}{\sqrt{x}-\sqrt{3x}}$
\item$\lim\limits_{x\to-1}\dfrac{\sqrt[3]{x+1}}{x+1}$
\item$\lim\limits_{x\to5}\dfrac{\sqrt{x-1}-2}{x-5}$
\end{enumerate}
\end{multicols}

\section*{Практикум 2. Первый замечательный предел}
\begin{multicols}{2}
\begin{enumerate}
\item$\lim\limits_{x\to0}\dfrac{\sin2x}{3x}$
\item$\lim\limits_{x\to0}\dfrac{\sin2x+\sin3x}{x}$
\item$\lim\limits_{x\to0}\dfrac{\sin3x}{\sin2x}$
\item$\lim\limits_{x\to0}\dfrac{\sqrt{x+4}-2}{\sin5x}$
\item$\lim\limits_{x\to0}\dfrac{1-\cos6x}{x\sin3x}$
\item$\lim\limits_{x\to0}\dfrac{\cos4x-\cos2x}{2x^2}$
\item$\lim\limits_{x\to0}\dfrac{1-\cos5x}{2x^2}$
\item$\lim\limits_{x\to0}\dfrac{\sin4x}{\sqrt{x+1}-1}$
\item$\lim\limits_{x\to0}\dfrac{\sin7x-\sin2x}{\sin x}$
\item$\lim\limits_{x\to0}\dfrac{1-\cos x}{x\sin x}$
\item$\lim\limits_{x\to0}\dfrac{\sin2x}{x}$
\item$\lim\limits_{x\to0}\dfrac{\sin^{2}(2x)}{x^2}$
\item$\lim\limits_{x\to0}\dfrac{\sin2x\sin x}{x^2}$
\item$\lim\limits_{x\to0}\dfrac{\sin3x}{3\tg x}$
\item$\lim\limits_{x\to0}\dfrac{\sin x}{\sin7x-\sin5x}$
\item$\lim\limits_{x\to0}\dfrac{\sin2x}{x^2+6x}$
\item$\lim\limits_{x\to0}\dfrac{\cos^{2}x-\cos^{2}(2x)}{3x^2}$
\item$\lim\limits_{x\to0}\left(\dfrac1{\sin2x}-\dfrac1{\tg2x}\right)$
\item$\lim\limits_{x\to0}\dfrac{\cos2x-1}{x\sin x}$
\item$\lim\limits_{x\to0}\dfrac{1-\cos3x}{\sin^{2}(7x)}$
\item$\lim\limits_{x\to0}\dfrac{\sqrt{1+\sin x}-\sqrt{1-\sin x}}{\tg x}$
\end{enumerate}
\end{multicols}

\section*{Практикум 3. Пределы на бесконечности и второй замечательный предел}
\begin{multicols}{2}
\begin{enumerate}
\item$\lim\limits_{x\to+\infty}\dfrac{x^4-2x^2+3}{3x^3-5}$
\item$\lim\limits_{x\to+\infty}\dfrac{3x^2+x+1}{5x^2-2x-3}$
\item$\lim\limits_{x\to+\infty}\dfrac{\sqrt{x^2+1}+x}{\sqrt[4]{x^3+x}-x}$
\item$\lim\limits_{x\to+\infty}\dfrac{(x+1)^3-(x-1)^3}{(x+1)^2+(x-1)^2}$
\item$\lim\limits_{x\to+\infty}(\sqrt{x^2+x-1}-\sqrt{x^2-x+1})$
\item$\lim\limits_{x\to0}(1+x)^{1/x}$
\item$\lim\limits_{x\to+\infty}\left(\dfrac{3x-2}{3x+5}\right)^{3x+1}$
\item$\lim\limits_{x\to+\infty}\left(\dfrac{1+x}{2+x}\right)^{1-x}$
\end{enumerate}
\end{multicols}
